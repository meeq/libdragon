29__attribute__((used))
30static inline int asset_buf_size(int size, int cmp_size, int margin, int *cmp_offset_dst)
31{
32 // add 8 because the assembly decompressors do writes up to 8 bytes out-of-bounds,
33 // that could overwrite the input data.
34 margin += 8;
35 int bufsize = size + margin;
36 int cmp_offset = bufsize - cmp_size;
37 // Align the source buffer to 4 bytes, so that we can use 32-bit loads (required by shrinkler).
38 // Notice that we need at least 2-byte alignment anyway, for DMA.
39 while (cmp_offset & 3) {
40 cmp_offset++;
41 bufsize++;
42 }
43 if(cmp_offset_dst) {
44 *cmp_offset_dst = cmp_offset;
45 }
46 if (bufsize & 15) {
47 // In case we need to call invalidate (see below), we need an aligned buffer
48 bufsize += 16 - (bufsize & 15);
49 }
50 return bufsize;
51}
