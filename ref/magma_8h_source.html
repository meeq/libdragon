662LIBDRAGON_PREVIEW_API
663inline uint8_t mg_culling_parms_to_rsp_state(const mg_culling_parms_t *culling)
664{
665 uint8_t cull_mode;
666 uint8_t is_front_cw;
667
668 switch (culling->cull_mode) {
669 case MG_CULL_MODE_NONE:
670 cull_mode = 2;
671 break;
672 case MG_CULL_MODE_BACK:
673 cull_mode = 1;
674 break;
675 case MG_CULL_MODE_FRONT:
676 cull_mode = 0;
677 break;
678 default:
679 assertf(0, "%d is not a valid cull mode!", culling->cull_mode);
680 cull_mode = 2;
681 }
682
683 switch (culling->front_face) {
684 case MG_FRONT_FACE_COUNTER_CLOCKWISE:
685 is_front_cw = 0;
686 break;
687 case MG_FRONT_FACE_CLOCKWISE:
688 is_front_cw = 1;
689 break;
690 default:
691 assertf(0, "%d is not a valid front face winding direction!", culling->front_face);
692 is_front_cw = 0;
693 }
694
695 // If the front face is clockwise, flip the cull mode.
696 // If the cull mode is NONE anyway, this has no effect because the final value is still > 1
697 return cull_mode ^ is_front_cw;
698}
