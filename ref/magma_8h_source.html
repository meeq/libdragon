624inline uint8_t mg_culling_parms_to_rsp_state(const mg_culling_parms_t *culling)
625{
626 uint8_t cull_mode;
627 uint8_t is_front_cw;
628
629 switch (culling->cull_mode) {
630 case MG_CULL_MODE_NONE:
631 cull_mode = 2;
632 break;
633 case MG_CULL_MODE_BACK:
634 cull_mode = 1;
635 break;
636 case MG_CULL_MODE_FRONT:
637 cull_mode = 0;
638 break;
639 default:
640 assertf(0, "%d is not a valid cull mode!", culling->cull_mode);
641 }
642
643 switch (culling->front_face) {
644 case MG_FRONT_FACE_COUNTER_CLOCKWISE:
645 is_front_cw = 0;
646 break;
647 case MG_FRONT_FACE_CLOCKWISE:
648 is_front_cw = 1;
649 break;
650 default:
651 assertf(0, "%d is not a valid front face winding direction!", culling->front_face);
652 }
653
654 // If the front face is clockwise, flip the cull mode.
655 // If the cull mode is NONE anyway, this has no effect because the final value is still > 1
656 return cull_mode ^ is_front_cw;
657}
