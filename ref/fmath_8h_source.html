107static inline float fm_truncf(float x) {
108 /* Notice that trunc.w.s is also emitted by the compiler when casting a
109 * float to int, but in this case we want a floating point result anyway,
110 * so it's useless to go back and forth a GPR. */
111 float yint, y;
112 __asm ("trunc.w.s %0,%1" : "=f"(yint) : "f"(x));
113 __asm ("cvt.s.w %0,%1" : "=f"(y) : "f"(yint));
114 return y;
115}

291 // The following macros contain a special-case: when called with constant
292 // arguments, they fall back to the standard math.h version. This allows
293 // the compiler to compute an accurate result at compile time.
294 // For instance, if we find `sinf(sqrtf(2.0f))` in the source code,
295 // we expect that to be resolved at compile-time with the maximum accuracy,
296 // so it's important to generate code that calls the standard C function
297 // which is then intrinsified by the compiler.
298 #define fmodf(x, y) ((__builtin_constant_p(x) && __builtin_constant_p(y)) ? fmodf(x,y) : fm_fmodf(x,y))
299 #define sinf(x) (__builtin_constant_p(x) ? sinf(x) : fm_sinf(x))
300 #define cosf(x) (__builtin_constant_p(x) ? cosf(x) : fm_cosf(x))
301 #define sincosf(x,s,c) (__builtin_constant_p(x) ? sincosf(x,s,c) : fm_sincosf(x,s,c))
302 #define atan2f(y, x) ((__builtin_constant_p(x) && __builtin_constant_p(y)) ? atan2f(y, x) : fm_atan2f(y, x))
303 #define expf(x) (__builtin_constant_p(x) ? expf(x) : fm_expf(x))
304#endif
