109static inline float fm_truncf(float x) {
110 /* Notice that trunc.w.s is also emitted by the compiler when casting a
111 * float to int, but in this case we want a floating point result anyway,
112 * so it's useless to go back and forth a GPR. */
113 float yint, y;
114 __asm ("trunc.w.s %0,%1" : "=f"(yint) : "f"(x));
115 __asm ("cvt.s.w %0,%1" : "=f"(y) : "f"(yint));
116 return y;
117}

317 // The following macros contain a special-case: when called with constant
318 // arguments, they fall back to the standard math.h version. This allows
319 // the compiler to compute an accurate result at compile time.
320 // For instance, if we find `sinf(sqrtf(2.0f))` in the source code,
321 // we expect that to be resolved at compile-time with the maximum accuracy,
322 // so it's important to generate code that calls the standard C function
323 // which is then intrinsified by the compiler.
324 #define fmodf(x, y) ((__builtin_constant_p(x) && __builtin_constant_p(y)) ? fmodf(x,y) : fm_fmodf(x,y))
325 #define sinf(x) (__builtin_constant_p(x) ? sinf(x) : fm_sinf(x))
326 #define cosf(x) (__builtin_constant_p(x) ? cosf(x) : fm_cosf(x))
327 #define sincosf(x,s,c) (__builtin_constant_p(x) ? sincosf(x,s,c) : fm_sincosf(x,s,c))
328 #define atan2f(y, x) ((__builtin_constant_p(x) && __builtin_constant_p(y)) ? atan2f(y, x) : fm_atan2f(y, x))
329 #define expf(x) (__builtin_constant_p(x) ? expf(x) : fm_expf(x))
330 #define exp2f(x) (__builtin_constant_p(x) ? exp2f(x) : fm_exp2f(x))
331#endif
