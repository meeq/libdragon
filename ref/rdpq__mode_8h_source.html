403inline void rdpq_mode_antialias(rdpq_antialias_t mode)
404{
405 // Just enable/disable SOM_AA_ENABLE. The RSP will then update the render mode
406 // which would trigger different other bits in SOM depending on the current mode.
407 __rdpq_mode_change_som(SOM_AA_ENABLE | SOMX_AA_REDUCED,
408 (mode ? SOM_AA_ENABLE : 0) | (mode == AA_REDUCED ? SOMX_AA_REDUCED : 0));
409}

481inline void rdpq_mode_combiner(rdpq_combiner_t comb) {
482 extern void __rdpq_fixup_mode(uint32_t cmd_id, uint32_t w0, uint32_t w1);
483 extern void __rdpq_fixup_mode4(uint32_t cmd_id, uint32_t w0, uint32_t w1, uint32_t w2, uint32_t w3);
484
485 if (comb & RDPQ_COMBINER_2PASS)
486 __rdpq_fixup_mode(RDPQ_CMD_SET_COMBINE_MODE_2PASS,
487 (comb >> 32) & 0x00FFFFFF,
488 comb & 0xFFFFFFFF);
489 else {
490 rdpq_combiner_t comb1_mask = RDPQ_COMB1_MASK;
491 if (((comb >> 0 ) & 7) == 1) comb1_mask ^= 1ull << 0;
492 if (((comb >> 3 ) & 7) == 1) comb1_mask ^= 1ull << 3;
493 if (((comb >> 6 ) & 7) == 1) comb1_mask ^= 1ull << 6;
494 if (((comb >> 18) & 7) == 1) comb1_mask ^= 1ull << 18;
495 if (((comb >> 21) & 7) == 1) comb1_mask ^= 1ull << 21;
496 if (((comb >> 24) & 7) == 1) comb1_mask ^= 1ull << 24;
497 if (((comb >> 32) & 31) == 1) comb1_mask ^= 1ull << 32;
498 if (((comb >> 37) & 15) == 1) comb1_mask ^= 1ull << 37;
499
500 __rdpq_fixup_mode4(RDPQ_CMD_SET_COMBINE_MODE_1PASS,
501 (comb >> 32) & 0x00FFFFFF,
502 comb & 0xFFFFFFFF,
503 (comb1_mask >> 32) & 0x00FFFFFF,
504 comb1_mask & 0xFFFFFFFF);
505 }
506}

925inline void __rdpq_mode_change_som(uint64_t mask, uint64_t val)
926{
927 // This is identical to #rdpq_change_other_modes_raw, but we also
928 // set bit 1<<15 in the first word. That flag tells the RSP code
929 // to recalculate the render mode, in addition to flipping the bits.
930 // #rdpq_change_other_modes_raw instead just changes the bits as
931 // you would expect from a raw API.
932 extern void __rdpq_fixup_mode3(uint32_t cmd_id, uint32_t w0, uint32_t w1, uint32_t w2);
933 if (mask >> 32)
934 __rdpq_fixup_mode3(RDPQ_CMD_MODIFY_OTHER_MODES, 0 | (1<<15), ~(mask >> 32), val >> 32);
935 if ((uint32_t)mask)
936 __rdpq_fixup_mode3(RDPQ_CMD_MODIFY_OTHER_MODES, 4 | (1<<15), ~(uint32_t)mask, (uint32_t)val);
937}
