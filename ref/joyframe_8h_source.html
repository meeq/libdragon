166 int idx = pkt[63]; \
167 pkt[idx++] = send_len; \
168 pkt[idx++] = recv_len; \
169 pkt[idx++] = cmd; \
170 uint8_t *send_ptr = &pkt[idx]; \
171 pkt[63] = idx + send_len + recv_len - 1; \
