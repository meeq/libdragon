127inline void __rdpq_tracking_state_reset(rdpq_tracking_t *state) {
128 *state = (rdpq_tracking_t){
129 // current autosync status is unknown because blocks can be
130 // played in any context. So assume the worst: all resources
131 // are being used. This will cause all SYNCs to be generated,
132 // which is the safest option.
133 .autosync = ~0,
134 // we don't know whether mode changes will be frozen or not
135 // when the block will play. Assume the worst (and thus
136 // do not optimize out mode changes).
137 .mode_freeze = false,
138 // we don't know the cycle type after we run the block
139 .cycle_type_known = 0,
140 .cycle_type_frozen = 0,
141 };
142}
