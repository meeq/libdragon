166LIBDRAGON_PREVIEW_API
167inline uint32_t emux_detect(int subcode)
168{
169 /*
170 * GCC can't easily feed the destination register number into a raw `.word`
171 * encoding. A simple MIPS64-friendly pattern is:
172 * 1) encode the instruction to write into a fixed scratch GPR (here: t0)
173 * 2) copy the scratch GPR into a GCC output register
174 */
175 const int REG_T0 = 8;
176 register uint64_t result asm("t0") = 0;
177
178 __asm__ __volatile__(
179 " .word %1\n"
180 : "+r"(result) : "i"(EMUX_XDETECT(REG_T0, subcode)) : "memory");
181
182 return result;
183}
