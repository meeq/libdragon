164inline uint32_t emux_detect(int subcode)
165{
166 /*
167 * GCC can't easily feed the destination register number into a raw `.word`
168 * encoding. A simple MIPS64-friendly pattern is:
169 * 1) encode the instruction to write into a fixed scratch GPR (here: t0)
170 * 2) copy the scratch GPR into a GCC output register
171 */
172 const int REG_T0 = 8;
173 register uint64_t result asm("t0") = 0;
174
175 __asm__ __volatile__(
176 " .word %1\n"
177 : "+r"(result) : "i"(EMUX_XDETECT(REG_T0, subcode)) : "memory");
178
179 return result;
180}
