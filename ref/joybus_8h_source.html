365inline void joybus_exec_cmd(
366 int port,
367 size_t send_len,
368 size_t recv_len,
369 const void *send_data,
370 void *recv_data
371)
372{
373 // Validate the desired Joybus port offset
374 assert((port >= 0) && (port < JOYBUS_PORT_COUNT));
375 // Ensure the send_len and recv_len fit in the Joybus operation block
376 assert((port + send_len + recv_len) < (JOYBUS_BLOCK_SIZE - 4));
377 // Allocate the Joybus operation block input and output buffers
378 uint8_t input[JOYBUS_BLOCK_SIZE] = {0};
379 uint8_t output[JOYBUS_BLOCK_SIZE] = {0};
380 // Skip commands on ports before the desired port offset
381 size_t i = port;
382 // Set the command metadata
383 input[i++] = send_len;
384 input[i++] = recv_len;
385 // Copy the send_data into the input buffer
386 memcpy(&input[i], send_data, send_len);
387 i += send_len + recv_len;
388 // Close out the Joybus operation block
389 input[i] = 0xFE;
390 input[sizeof(input) - 1] = 0x01;
391 // Execute the Joybus operation synchronously
392 joybus_exec(input, output);
393 // Copy recv_data from the output buffer
394 memcpy(recv_data, &output[i - recv_len], recv_len);
395}
