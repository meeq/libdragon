372inline void joybus_exec_cmd(
373 int port,
374 size_t send_len,
375 size_t recv_len,
376 const void *send_data,
377 void *recv_data
378)
379{
380 // Validate the desired Joybus port offset
381 assert((port >= 0) && (port < JOYBUS_PORT_COUNT));
382 // Ensure the send_len and recv_len fit in the Joybus operation block
383 assert((port + send_len + recv_len) < (JOYBUS_BLOCK_SIZE - 4));
384 // Allocate the Joybus operation block input and output buffers
385 uint8_t input[JOYBUS_BLOCK_SIZE] = {0};
386 uint8_t output[JOYBUS_BLOCK_SIZE] = {0};
387 // Skip commands on ports before the desired port offset
388 size_t i = port;
389 // Set the command metadata
390 input[i++] = send_len;
391 input[i++] = recv_len;
392 // Copy the send_data into the input buffer
393 memcpy(&input[i], send_data, send_len);
394 i += send_len + recv_len;
395 // Close out the Joybus operation block
396 input[i] = 0xFE;
397 input[sizeof(input) - 1] = 0x01;
398 // Execute the Joybus operation synchronously
399 joybus_exec(input, output);
400 // Copy recv_data from the output buffer
401 memcpy(recv_data, &output[i - recv_len], recv_len);
402}
