568inline rspq_write_t rspq_write_begin(uint32_t ovl_id, uint32_t cmd_id, int size) {
569 extern volatile uint32_t *rspq_cur_pointer, *rspq_cur_sentinel;
570 extern void rspq_next_buffer(void);
571
572 assertf(size <= RSPQ_MAX_COMMAND_SIZE, "The maximum command size is %d!", RSPQ_MAX_COMMAND_SIZE);
573
574 if (__builtin_expect(rspq_cur_pointer > rspq_cur_sentinel - size, 0))
575 rspq_next_buffer();
576
577 volatile uint32_t *cur = rspq_cur_pointer;
578 rspq_cur_pointer += size;
579
580 return (rspq_write_t){
581 .first_word = ovl_id + (cmd_id<<24),
582 .pointer = cur + 1,
583 .first = cur,
584 .is_first = 1
