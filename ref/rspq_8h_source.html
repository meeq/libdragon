572inline rspq_write_t rspq_write_begin(uint32_t ovl_id, uint32_t cmd_id, int size) {
573 extern volatile uint32_t *rspq_cur_pointer, *rspq_cur_sentinel;
574 extern void rspq_next_buffer(void);
575
576 assertf(size <= RSPQ_MAX_COMMAND_SIZE, "The maximum command size is %d!", RSPQ_MAX_COMMAND_SIZE);
577
578 if (__builtin_expect(rspq_cur_pointer > rspq_cur_sentinel - size, 0))
579 rspq_next_buffer();
580
581 volatile uint32_t *cur = rspq_cur_pointer;
582 rspq_cur_pointer += size;
583
584 return (rspq_write_t){
585 .first_word = ovl_id + (cmd_id<<24),
586 .pointer = cur + 1,
587 .first = cur,
588 .is_first = 1
589 };
590}
