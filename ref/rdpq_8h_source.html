776inline void rdpq_set_tile(rdpq_tile_t tile,
777 tex_format_t format,
778 int32_t tmem_addr,
779 uint16_t tmem_pitch,
780 const rdpq_tileparms_t *parms)
781{
782 static const rdpq_tileparms_t default_parms = {0};
783 if (!parms) parms = &default_parms;
784 else {
785 assertf(parms->s.shift >= -5 && parms->s.shift <= 10, "invalid s shift %d: must be in [-5..10]", parms->s.shift);
786 assertf(parms->t.shift >= -5 && parms->t.shift <= 10, "invalid t shift %d: must be in [-5..10]", parms->t.shift);
787 assertf(parms->palette < 16, "invalid palette %d: must be in [0..15]", parms->palette);
788 }
789 bool fixup = false;
790 bool reuse = false;
791 uint32_t cmd_id = RDPQ_CMD_SET_TILE;
792 if (tmem_addr & (RDPQ_AUTOTMEM | RDPQ_AUTOTMEM_REUSE(0))) {
793 cmd_id = RDPQ_CMD_AUTOTMEM_SET_TILE;
794 reuse = (tmem_addr & RDPQ_AUTOTMEM_REUSE(0)) != 0;
795 fixup = true;
796 tmem_addr &= ~(RDPQ_AUTOTMEM | RDPQ_AUTOTMEM_REUSE(0));
797 } else {
798 assertf((tmem_addr % 8) == 0, "invalid tmem_addr %ld: must be multiple of 8", tmem_addr);
799 tmem_addr /= 8;
800 }
801 assertf((tmem_pitch % 8) == 0, "invalid tmem_pitch %d: must be multiple of 8", tmem_pitch);
802 extern void __rdpq_write8_syncchange(uint32_t, uint32_t, uint32_t, uint32_t);
803 extern void __rdpq_fixup_write8_syncchange(uint32_t, uint32_t, uint32_t, uint32_t);
804 (fixup ? __rdpq_fixup_write8_syncchange : __rdpq_write8_syncchange)(cmd_id,
805 _carg(format, 0x1F, 19) | _carg(reuse, 0x1, 18) | _carg(tmem_pitch/8, 0x1FF, 9) | _carg(tmem_addr, 0x1FF, 0),
806 _carg(tile, 0x7, 24) | _carg(parms->palette, 0xF, 20) |
807 _carg(parms->t.clamp | (parms->t.mask == 0), 0x1, 19) | _carg(parms->t.mirror, 0x1, 18) | _carg(parms->t.mask, 0xF, 14) | _carg(parms->t.shift, 0xF, 10) |
808 _carg(parms->s.clamp | (parms->s.mask == 0), 0x1, 9) | _carg(parms->s.mirror, 0x1, 8) | _carg(parms->s.mask, 0xF, 4) | _carg(parms->s.shift, 0xF, 0),
809 AUTOSYNC_TILE(tile));
810}
