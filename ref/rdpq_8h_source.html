408 uint16_t wg = fwg * 255.0f;

780inline void rdpq_set_tile(rdpq_tile_t tile,
781 tex_format_t format,
782 int32_t tmem_addr,
783 uint16_t tmem_pitch,
784 const rdpq_tileparms_t *parms)
785{
786 static const rdpq_tileparms_t default_parms = {0};
787 if (!parms) parms = &default_parms;
788 else {
789 assertf(parms->s.shift >= -5 && parms->s.shift <= 10, "invalid s shift %d: must be in [-5..10]", parms->s.shift);
790 assertf(parms->t.shift >= -5 && parms->t.shift <= 10, "invalid t shift %d: must be in [-5..10]", parms->t.shift);
791 assertf(parms->palette < 16, "invalid palette %d: must be in [0..15]", parms->palette);
792 }
793 bool fixup = false;
794 bool reuse = false;
795 uint32_t cmd_id = RDPQ_CMD_SET_TILE;
796 if (tmem_addr & (RDPQ_AUTOTMEM | RDPQ_AUTOTMEM_REUSE(0))) {
797 cmd_id = RDPQ_CMD_AUTOTMEM_SET_TILE;
798 reuse = (tmem_addr & RDPQ_AUTOTMEM_REUSE(0)) != 0;
799 fixup = true;
800 tmem_addr &= ~(RDPQ_AUTOTMEM | RDPQ_AUTOTMEM_REUSE(0));
801 } else {
802 assertf((tmem_addr % 8) == 0, "invalid tmem_addr %ld: must be multiple of 8", tmem_addr);
803 tmem_addr /= 8;
804 }
805 assertf((tmem_pitch % 8) == 0, "invalid tmem_pitch %d: must be multiple of 8", tmem_pitch);
806 extern void __rdpq_write8_syncchange(uint32_t, uint32_t, uint32_t, uint32_t);
807 extern void __rdpq_fixup_write8_syncchange(uint32_t, uint32_t, uint32_t, uint32_t);
808 (fixup ? __rdpq_fixup_write8_syncchange : __rdpq_write8_syncchange)(cmd_id,
809 _carg(format, 0x1F, 19) | _carg(reuse, 0x1, 18) | _carg(tmem_pitch/8, 0x1FF, 9) | _carg(tmem_addr, 0x1FF, 0),
810 _carg(tile, 0x7, 24) | _carg(parms->palette, 0xF, 20) |
811 _carg(parms->t.clamp | (parms->t.mask == 0), 0x1, 19) | _carg(parms->t.mirror, 0x1, 18) | _carg(parms->t.mask, 0xF, 14) | _carg(parms->t.shift, 0xF, 10) |
812 _carg(parms->s.clamp | (parms->s.mask == 0), 0x1, 9) | _carg(parms->s.mirror, 0x1, 8) | _carg(parms->s.mask, 0xF, 4) | _carg(parms->s.shift, 0xF, 0),
813 AUTOSYNC_TILE(tile));
814}
